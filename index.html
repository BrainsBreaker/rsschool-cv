        def find_link(A, B, k):
            res = requests.get(A)

                return False
            else:
                for serch in re.finditer(r'<a.*href=\"', res.text):

                    i = serch.span()[1]
                    while res.text[i] != '\"':
                        C1 += res.text[i]
                        i += 1
                    B2 = B.replace('stepik.org', 'stepic.org')
                    C2 = C1.replace('stepik.org', 'stepic.org')
                    if C2 == B2 and k == 1:
                        return True
                    elif k<1:
                        if find_link(C1, B, k+1) or find_link(C2, B, k+1):
                            return True

            A = input()
